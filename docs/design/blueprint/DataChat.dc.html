SELECT region,
       date_trunc('day', bestellt_am) AS tag,
       sum(betrag_netto)               AS umsatz
FROM   bestellungen
WHERE  bestellt_am >= now() - interval '30 days'
  AND  status <> 'storniert'
GROUP BY region, tag
ORDER BY tag;
